# ─── 一键全自动 ───
# 在 Windows PowerShell 中运行:
python auto_run.py
# 等效: python auto_run.py --package com.google.android.GoogleCamera
# 注意: Windows 上请用 python, python3 是 Microsoft Store 桩, 不可用

# 分步运行 (跳过某些阶段):
python auto_run.py --skip-build           # 跳过编译, 只部署+采集+分析
python auto_run.py --only-analyze         # 用已有的数据仅重跑分析
python auto_run.py --duration 60          # 采集60秒后超时自动停止

# ─── 手动编译 (仅需一次, 除非改 C 代码) ───
cd ebpf && make
# 产物: build/camera_ebpf_android

# ─── 手动部署 eBPF ───
adb push ./ebpf/build/camera_ebpf_android /data/local/tmp/
adb shell 
su
cd /data/local/tmp/
chmod +x camera_ebpf_android

# 务必先启动 Perfetto, 再启动 eBPF!
# 顺序: 1) Perfetto 先跑 → 2) eBPF 再跑 → 3) 操作 App
# 原因: eBPF 的 RingBuffer 只有 512KB, 先跑会提前填满,
#       导致卡顿发生时无空闲缓冲可用。

# ─── 手动抓取 Perfetto Trace (先执行) ───
adb push perfetto_camera.pbtx /data/local/tmp/
adb shell "cat /data/local/tmp/perfetto_camera.pbtx | perfetto --txt -c - -o /data/misc/perfetto-traces/camera_jank.perfetto"

# ─── 再启动 eBPF 采集 (后执行) ───
# 新功能: -u <uid> 可让内核侧过滤 futex 事件, 减少 90% 系统调用事件量
# 如何获取 UID: adb shell stat -c '%u' /data/data/<package>
./camera_ebpf_android -q -u 10162
# Ctrl+C 停止后生成 sched_events.csv 和 binder_futex_events.csv


# ─── 手动拉取产物 ───
adb pull /data/misc/perfetto-traces/camera_jank.perfetto ./perfetto/
# CSV 在 output/raw/ 下
adb pull /data/local/tmp/sched_events.csv ./output/raw/
adb pull /data/local/tmp/binder_futex_events.csv ./output/raw/


# ─── 手动分析流程 ───
cd perfetto
python3 parse_trace.py camera_jank.perfetto com.google.android.GoogleCamera

cd ../ebpf
# 一键分析: sched延迟 + binder依赖边 + futex活动 + Critical Path Graph + CriticalScore
python3 analyze_delays.py --json ../output/analysis/ebpf_target_windows.json --csv ../output/raw/sched_events.csv --binder ../output/raw/binder_futex_events.csv

# 也可单独运行图构建:
# python3 critical_path.py --delay ../output/analysis/delay_analysis_result.json --window ../output/analysis/ebpf_target_windows.json

# ─── 产物 ───
# output/analysis/ebpf_target_windows.json  → 卡顿帧时间窗口
# output/analysis/delay_analysis_result.json → 每帧的 sched/binder/futex 详细数据
# output/analysis/critical_path_graph.json  → 图结构 + Top-K CriticalScore 排名
# output/reports/report_*.md                → 可读的分析报告文档
